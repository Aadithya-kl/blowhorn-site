{{- /* Resolve one integrations.json platform row. Name, logo and actions come from hugo.Data.platforms (data/platforms/.yaml) plus data/actions.json for display names. A platform with no YAML keeps its own name and logo from the row and lists no actions. resolved is each action-resolve.html map (platform behavior and limits), for the action matrix. */ -}} {{- $row := . -}} {{- $yaml := index hugo.Data.platforms $row.id -}} {{- $name := $row.name -}} {{- $logo := $row.logo -}} {{- $actions := slice -}} {{- $actionIds := slice -}} {{- $actionLinks := slice -}} {{- $resolved := slice -}} {{- if $yaml -}} {{- $name = $yaml.name -}} {{- $logo = $yaml.logo -}} {{- range $yaml.actions }} {{- $a := partial "action-resolve.html" . -}} {{- $actionIds = $actionIds | append $a.id -}} {{- $actions = $actions | append $a.name -}} {{- $actionLinks = $actionLinks | append (dict "id" $a.id "name" $a.name "slug" $a.slug) -}} {{- $resolved = $resolved | append $a -}} {{- end -}} {{- else if or (not $name) (not $logo) -}} {{- errorf "integration-row: platform %q has no data/platforms/%s.yaml and no name or logo in data/integrations.json" $row.id $row.id -}} {{- end -}} {{- return dict "id" $row.id "status" $row.status "posts" $row.posts "docs" $row.docs "viaLabel" $row.viaLabel "exampleLink" $row.exampleLink "demoMeta" $row.demoMeta "name" $name "logo" $logo "actions" $actions "actionIds" $actionIds "actionLinks" $actionLinks "resolved" $resolved -}}
